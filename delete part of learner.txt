def learner_edit(request,id):
e=Learners.objects.get(id=id)
if request.method=="POST":
  f=request.POST["first_name"]
  l=request.POST["last_name"]
  p=request.POST["place"]
  a=request.POST["address"]
  
  e.first_name=f
  e.last_name=l
  e.place=p
  e.address=a
  e.save()
  return redirect(learner_view)
else:
    return render(request,"learner_edit.html",{"edit":e})
